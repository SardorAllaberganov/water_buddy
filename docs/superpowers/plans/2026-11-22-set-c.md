## Shared tooling — adding strings

Catalogues are appended to, never re-sorted: their key order is insertion order, and the file round-trips
byte-identically through `json.dumps(…, indent=2, ensure_ascii=False) + "\n"` (measured 2026-10-30; check again
in Step 1 below). Write this once to the session's scratchpad as `add_strings.py`:

```python
# python3 -I add_strings.py <catalogue.xcstrings> <keys.json>
import json, sys
path, spec = sys.argv[1], sys.argv[2]
src = open(path, encoding="utf-8").read()
cat = json.loads(src)
assert json.dumps(cat, indent=2, ensure_ascii=False) + "\n" == src, "the catalogue does not round-trip; stop"
new = json.load(open(spec, encoding="utf-8"))
for key, langs in new.items():
    assert key not in cat["strings"], f"{key!r} already exists"
    locs = {}
    for lang in ("en", "ru", "uz"):
        value = langs[lang]
        if isinstance(value, str):
            locs[lang] = {"stringUnit": {"state": "translated", "value": value}}
        else:
            locs[lang] = {"variations": {"plural": {form: {"stringUnit": {"state": "translated", "value": v}}
                                                   for form, v in sorted(value.items())}}}
    cat["strings"][key] = {"extractionState": "manual", "localizations": locs}
open(path, "w", encoding="utf-8").write(json.dumps(cat, indent=2, ensure_ascii=False) + "\n")
print(f"added {len(new)} keys")
```

Each task that adds strings writes its own `keys-<task>.json` beside it and runs
`python3 -I <scratchpad>/add_strings.py WaterBuddy/Localizable.xcstrings <scratchpad>/keys-<task>.json`.
`LocalizationTests.everyDrawnStringIsTranslatedUnlessDeliberatelyNot` then holds every new key in every language;
none of set C's is deliberately English.

/// The seven icons a vessel may wear — the standard three's and four more — in the order the editor offers
/// them (spec `2026-11-22-set-c-design.md` D13).
///
/// **Stored by code**, the raw value, never by symbol name: a symbol Apple renames changes one `case` here and
/// no stored library. Every symbol exists on iOS 17 and watchOS 26 — read from SF Symbols' own availability
/// table; `cup.and.heat.waves.fill` needs iOS 18 and is left out. `VesselGlyphTests` proves each resolves,
/// because an unknown symbol draws nothing at all. Here, beside `vesselSlots`, because the widget's snapshot
/// and the watch read it and this file is in all three exception sets (rule `15-project`).
nonisolated enum VesselGlyph: String, CaseIterable, Codable, Sendable {
    case cup, glass, bottle, wineGlass, takeaway, flask, drop

    /// The SF Symbol drawn for it.
    var symbol: String {
        switch self {
        case .cup: "cup.and.saucer.fill"
        case .glass: "mug.fill"
        case .bottle: "waterbottle.fill"
        case .wineGlass: "wineglass.fill"
        case .takeaway: "takeoutbag.and.cup.and.straw.fill"
        case .flask: "flask.fill"
        case .drop: "drop.fill"
        }
    }

    /// What VoiceOver calls the icon in the vessel editor — the icon, not a vessel's name.
    var spokenKey: String {
        switch self {
        case .cup: "Cup"
        case .glass: "Mug"
        case .bottle: "Bottle"
        case .wineGlass: "Wine glass"
        case .takeaway: "Takeaway cup"
        case .flask: "Flask"
        case .drop: "Drop"
        }
    }

    /// The translated name a standard vessel wears: only the standard three's glyphs have one. A user's vessel
    /// always carries the name they typed.
    var standardNameKey: String? {
        switch self {
        case .cup: "Cup"
        case .glass: "Glass"
        case .bottle: "Bottle"
        case .wineGlass, .takeaway, .flask, .drop: nil
        }
    }

    /// The standard three's glyphs, in their own places: Cup, Glass, Bottle.
    static let standardThree: [VesselGlyph] = [.cup, .glass, .bottle]
}

/// The three quick-add vessels' presentation — names and SF Symbols, smallest first: Cup, Glass,
/// Bottle. …(the existing DocC, unchanged)…
///
/// **Since set C, the standard three's presentation alone**, built from ``VesselGlyph``: what Home, the watch and

}

// MARK: - The vessel library

/// One vessel the user can pour from — one of the standard three, or one of their own (spec
/// `2026-11-22-set-c-design.md` §3.6).
///
/// **The standard three keep their glyph and their translated name**, and are never deleted: `name` is `nil` for
/// them and their id is one of ``standardIDs``. A user's vessel has a UUID's string for an id and the name they
/// typed. `nonisolated`, because the widget's snapshot and the mirror composer read it (rule `43-concurrency`).
nonisolated struct Vessel: Codable, Sendable, Equatable, Identifiable {
    let id: String
    let glyph: VesselGlyph
    /// `nil` for a standard vessel, whose name is translated; the user's own words otherwise.
    let name: String?
    let amount: Int

    /// The standard three's ids, in their own places: Cup, Glass, Bottle.
    static let standardIDs = ["cup", "glass", "bottle"]

    /// The longest name, in `Character`s (D10) — what Home's caption and the watch's label can carry on one line.
    static let nameLimit = 20

    var isStandard: Bool { Self.standardIDs.contains(id) }

    /// The standard vessel of `slot` — 0 Cup, 1 Glass, 2 Bottle — wearing `amount`.
    static func standard(_ slot: Int, amount: Int) -> Vessel {
        Vessel(id: standardIDs[slot], glyph: VesselGlyph.standardThree[slot], name: nil, amount: amount)
    }

    /// `typed`, trimmed, if it can be a name — one to ``nameLimit`` characters, on one line — or `nil`.
    static func validName(_ typed: String) -> String? {
        let trimmed = typed.trimmingCharacters(in: .whitespacesAndNewlines)
        guard (1...nameLimit).contains(trimmed.count), !trimmed.contains(where: \.isNewline) else { return nil }
        return trimmed
    }

    /// The name drawn: the user's own words, or the standard vessel's, translated in `bundle`.
    func name(in bundle: Bundle) -> String {
        if let name { return name }
        let key = glyph.standardNameKey ?? "Glass"
        return bundle.localizedString(forKey: key, value: key, table: nil)
    }

    /// Whether this vessel keeps every rule a stored one must: a standard one wears its own glyph and no name;
    /// one of the user's has a valid name; every amount is in `1...maximumDailyIntake`.
    var isWellFormed: Bool {
        guard (1...DataManager.maximumDailyIntake).contains(amount) else { return false }
        if let index = Self.standardIDs.firstIndex(of: id) {
            return name == nil && glyph == VesselGlyph.standardThree[index]
        }
        guard let name, Self.validName(name) == name else { return false }
        return true
    }
}

/// Every vessel the user has: Home's three, positional, and up to nine more (spec `2026-11-22-set-c-design.md`
/// §3.6). Stored whole under `Key.vessels`; **validated on the way out and discarded whole** on any anomaly, as
/// the servings triple is (rule `20-state`).
///
/// Every change below is a pure function returning the next library, or `nil` when the change is refused — so
/// `DataManager` can never write a library this type would refuse to read.
nonisolated struct VesselLibrary: Codable, Sendable, Equatable {
    /// Home's three, positional: `home[DataManager.usualSlot]` is the Glass every one-tap door logs.
    let home: [Vessel]
    /// The rest, at most ``moreLimit``, in the order they were added or moved.
    let more: [Vessel]

    /// Nine more beside Home's three: twelve in all (D10).
    static let moreLimit = 9

    /// A library that was never made: the standard three in their own places, wearing `amounts`.
    static func standard(amounts: [Int]) -> VesselLibrary {
        VesselLibrary(
            home: (0..<3).map { Vessel.standard($0, amount: amounts.indices.contains($0) ? amounts[$0] : DataManager.defaultServings[$0]) },
            more: []
        )
    }

    /// Home's three amounts — what `Key.servings` holds, and every one-tap door reads.
    var amounts: [Int] { home.map(\.amount) }

    var all: [Vessel] { home + more }

    func vessel(id: String) -> Vessel? { all.first { $0.id == id } }

    /// Every rule a stored library keeps: exactly three on Home, at most nine more, every id once, the standard
    /// three present, and every vessel well formed.
    var isValid: Bool {
        guard home.count == 3, more.count <= Self.moreLimit else { return false }
        let ids = all.map(\.id)
        guard Set(ids).count == ids.count, Vessel.standardIDs.allSatisfy(ids.contains) else { return false }
        return all.allSatisfy(\.isWellFormed)
    }

    /// Whether Home holds Cup, Glass and Bottle in their own places — when *Use the standard vessels* has nothing to do.
    var isStandardOnHome: Bool { home.map(\.id) == Vessel.standardIDs }

    /// A vessel of the user's own, added to *More vessels* under `id`.
    func adding(glyph: VesselGlyph, name typed: String, amount: Int, id: String) -> VesselLibrary? {
        guard more.count < Self.moreLimit, let name = Vessel.validName(typed) else { return nil }
        let next = VesselLibrary(home: home, more: more + [Vessel(id: id, glyph: glyph, name: name, amount: amount)])
        return next.isValid ? next : nil
    }

    /// `id` changed in place. A standard vessel changes its amount alone; one of the user's, all three.
    func updating(id: String, glyph: VesselGlyph, name typed: String?, amount: Int) -> VesselLibrary? {
        guard let old = vessel(id: id) else { return nil }
        let updated: Vessel
        if old.isStandard {
            guard glyph == old.glyph, typed == nil else { return nil }
            updated = Vessel(id: id, glyph: glyph, name: nil, amount: amount)
        } else {
            guard let typed, let name = Vessel.validName(typed) else { return nil }
            updated = Vessel(id: id, glyph: glyph, name: name, amount: amount)
        }
        let swap = { (vessel: Vessel) in vessel.id == id ? updated : vessel }
        let next = VesselLibrary(home: home.map(swap), more: more.map(swap))
        return next.isValid ? next : nil
    }

    /// One of the user's vessels, removed from *More vessels*. Never a standard vessel, never one on Home.
    func deleting(id: String) -> VesselLibrary? {
        guard let index = more.firstIndex(where: { $0.id == id }), !more[index].isStandard else { return nil }
        var rest = more
        rest.remove(at: index)
        return VesselLibrary(home: home, more: rest)
    }

    /// `id` put in Home's `slot`. From *More vessels* it trades places with the vessel standing there; from
    /// another slot on Home, the two slots trade.
    func puttingOnHome(id: String, slot: Int) -> VesselLibrary? {
        guard home.indices.contains(slot) else { return nil }
        var home = self.home
        var more = self.more
        if let from = home.firstIndex(where: { $0.id == id }) {
            home.swapAt(from, slot)
        } else if let from = more.firstIndex(where: { $0.id == id }) {
            (home[slot], more[from]) = (more[from], home[slot])
        } else {
            return nil
        }
        return VesselLibrary(home: home, more: more)
    }

    /// Cup, Glass and Bottle back in their own places; whatever stood there joins *More vessels* after the rest.
    /// Nothing is deleted — the free way back for a lapsed holder (D5).
    func standardOnHome() -> VesselLibrary {
        VesselLibrary(
            home: Vessel.standardIDs.compactMap(vessel(id:)),
            more: more.filter { !$0.isStandard } + home.filter { !$0.isStandard }
        )
    }

    /// Home's three amounts replaced together — the `servings` setter's path while a library exists.
    func withHomeAmounts(_ amounts: [Int]) -> VesselLibrary? {
        guard amounts.count == home.count else { return nil }
        let next = VesselLibrary(
            home: zip(home, amounts).map { Vessel(id: $0.id, glyph: $0.glyph, name: $0.name, amount: $1) },
            more: more
        )
        return next.isValid ? next : nil
    }
}

/// The theme the app draws in — Aurora until the user chooses another (spec `2026-11-22-set-c-design.md` D9).
///
/// **Stored as ``AppLanguage`` is**: its code under `Key.theme`, and Aurora as the key's absence, so "never chose"
/// and "chose Aurora" are one state. Beside `AppLanguage` for its reason: the snapshot and the mirror carry it, and
/// no seventh shared file is added (rule `40-widget`). Its palette is ``palette`` (`WaterSurface.swift`).
nonisolated enum AppTheme: String, CaseIterable, Sendable, Identifiable {
    case aurora, lagoon, dusk, midnight, bloom

    var id: String { rawValue }

    /// What goes in the suite: `nil` for Aurora, which is why the setter removes the key.
    var code: String? { self == .aurora ? nil : rawValue }

    /// Reads a stored code — Aurora for none, and for one this build does not know, said under `#if DEBUG`
    /// (rule `75-diagnostics`). `"aurora"` is accepted and is Aurora.
    init(code: String?) {
        guard let code, let match = AppTheme(rawValue: code) else {
            #if DEBUG
            if let code, !code.isEmpty {
                print("WaterBuddy: unrecognised theme code — drawing Aurora instead")
            }
            #endif
            self = .aurora
            return
        }
        self = match
    }

    /// The theme's name, as the Appearance card draws it and VoiceOver speaks it.
    var nameKey: String {
        switch self {
        case .aurora: "Aurora"
        case .lagoon: "Lagoon"
        case .dusk: "Dusk"
        case .midnight: "Midnight"

    var r: CGFloat = 0, g: CGFloat = 0, b: CGFloat = 0, a: CGFloat = 0

private func luminance(_ c: (r: Double, g: Double, b: Double)) -> Double {
    func lin(_ v: Double) -> Double { v <= 0.04045 ? v / 12.92 : pow((v + 0.055) / 1.055, 2.4) }
    return 0.2126 * lin(c.r) + 0.7152 * lin(c.g) + 0.0722 * lin(c.b)
}

    let (la, lb) = (luminance(a), luminance(b))
    return (max(la, lb) + 0.05) / (min(la, lb) + 0.05)
}

// MARK: - The theme census

/// Every screen `docs/DESIGN.md` measures, in one theme, for `Tools/MeasureContrast.swift` (spec
/// `2026-11-22-set-c-design.md` §3.9). Driven only by `Tools/CaptureThemeCensus.sh`, one theme a run; the gate skips
/// this class (rule `85-testing`). The theme arrives as `TEST_RUNNER_WATERBUDDY_THEME`, which `xcodebuild` hands the
/// runner as `WATERBUDDY_THEME`.
extension AppStoreScreenshotUITests {

    /// Seconds between frames, and frames per screen that stands on the aurora: a minute, the swing of all three
    /// lights at least once (`tasks/lessons.md` 2759).
    private static let frameGap: TimeInterval = 3
    private static let framesOnTheAurora = 20

    func testCaptureThemeCensus() throws {
        let env = ProcessInfo.processInfo.environment
        let theme = try XCTUnwrap(env["WATERBUDDY_THEME"], "run through CaptureThemeCensus.sh")
        // One group of screens a run, so each command ends inside the 600-s limit (the driver's groups).
        let screens = Set((env["WATERBUDDY_SCREENS"] ?? "").split(separator: ",").map(String.init))
        let wanted = { (screen: String) in screens.isEmpty || screens.contains(screen) }

        if wanted("paywall") {
            // The paywall, as a free user meets it.
            let free = XCUIApplication()
            free.launchArguments = ["-WaterBuddyFreshStart", "YES", "-WaterBuddyStoreSample", "YES", "-WaterBuddyPremium", "YES",
                                    "-WaterBuddyTheme", theme]
            free.launch()
            completeWelcomeIfShown(free)
            free.buttons["Settings"].tap()
            free.buttons["See Premium"].tap()
            try sweep(free, screen: "paywall", theme: theme, frames: Self.framesOnTheAurora)
            free.terminate()
        }

        let holderScreens = ["home", "history-week", "history-month", "history-insights", "settings", "settings-lower", "library"]
        guard holderScreens.contains(where: wanted) else { return }
        // Everything else, as a holder. The walk always goes the whole way; only the wanted screens are swept.
        let app = XCUIApplication()
        app.launchArguments = ["-WaterBuddyFreshStart", "YES", "-WaterBuddyStoreSample", "YES", "-WaterBuddyPremiumHeld", "YES",
                               "-WaterBuddyPremium", "YES", "-WaterBuddyTheme", theme]
        app.launch()
        completeWelcomeIfShown(app)
        if wanted("home") { try sweep(app, screen: "home", theme: theme, frames: Self.framesOnTheAurora) }
        app.buttons["History"].tap()
        if wanted("history-week") { try sweep(app, screen: "history-week", theme: theme, frames: Self.framesOnTheAurora) }
        for mode in ["Month", "Insights"] {
            app.buttons[mode].tap()
            let screen = "history-\(mode.lowercased())"
            if wanted(screen) { try sweep(app, screen: screen, theme: theme, frames: Self.framesOnTheAurora) }
        }
        app.buttons["Settings"].tap()
        if wanted("settings") { try sweep(app, screen: "settings", theme: theme, frames: Self.framesOnTheAurora) }
        app.swipeUp()
        if wanted("settings-lower") { try sweep(app, screen: "settings-lower", theme: theme, frames: Self.framesOnTheAurora) }
        if wanted("library") {
            app.buttons["Your vessels"].tap()
            try sweep(app, screen: "library", theme: theme, frames: Self.framesOnTheAurora)
        }
    }

    /// `frames` screenshots `frameGap` apart, each attached, and the frames of every element a pair names, once.
    private func sweep(_ app: XCUIApplication, screen: String, theme: String, frames: Int) throws {
        sleep(2)   // the screen's arrival settles first
        var elements: [String: [Double]] = [:]
        for element in app.descendants(matching: .any).allElementsBoundByIndex where element.exists && !element.label.isEmpty {
            let f = element.frame
            elements[element.label] = [f.minX, f.minY, f.width, f.height]
        }
        let json = XCTAttachment(data: try JSONSerialization.data(withJSONObject: elements, options: [.sortedKeys]),
                                 uniformTypeIdentifier: "public.json")
        json.name = "\(theme)-\(screen)-frames.json"
        json.lifetime = .keepAlways
        add(json)
        for n in 0..<frames {
            let shot = XCTAttachment(screenshot: XCUIScreen.main.screenshot())
            shot.name = String(format: "%@-%@-%02d", theme, screen, n)
            shot.lifetime = .keepAlways
            add(shot)
            if n < frames - 1 { Thread.sleep(forTimeInterval: Self.frameGap) }
        }
    }
}

`Tools/MeasureContrast.swift` (run with `swift`; in no target, for `GenerateAppIcon.swift`'s reason):

```swift
//
//  MeasureContrast.swift
//  WaterBuddy — tooling, NOT a build input (spec 2026-11-22-set-c-design.md §3.9)
//
//  Run:  swift Tools/MeasureContrast.swift build/themes/<theme>-<runtime> Tools/theme-pairs.json
//
//  WCAG relative luminance, sRGB, at the capture's 3× pixels. For each pair and each frame: the element's box from
//  the census's frames, the ground on a ring around it — every other element's box masked out, dilated 3 px — and
//  the worst (lightest-ground) pixel. A pair's figure is its worst frame. The method `docs/DESIGN.md` records,
//  kept in one place this time.
//

import CoreGraphics
import Foundation
import ImageIO

struct Pair: Decodable { let screen, element, kind: String; let floor: Double; let ink: String
                         let inkOpacity: Double?; let yardstick: String; let note: String? }

func luminance(_ r: Double, _ g: Double, _ b: Double) -> Double {
    func lin(_ v: Double) -> Double { v <= 0.04045 ? v / 12.92 : pow((v + 0.055) / 1.055, 2.4) }
    return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b)
}
func ratio(_ a: Double, _ b: Double) -> Double { (max(a, b) + 0.05) / (min(a, b) + 0.05) }

struct Pixels {
    let width: Int, height: Int, data: [UInt8]
    init?(_ url: URL) {
        guard let source = CGImageSourceCreateWithURL(url as CFURL, nil),
              let image = CGImageSourceCreateImageAtIndex(source, 0, nil) else { return nil }
        width = image.width; height = image.height
        var bytes = [UInt8](repeating: 0, count: width * height * 4)
        let space = CGColorSpace(name: CGColorSpace.sRGB)!
        guard let ctx = CGContext(data: &bytes, width: width, height: height, bitsPerComponent: 8, bytesPerRow: width * 4,
                                  space: space, bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return nil }
        ctx.draw(image, in: CGRect(x: 0, y: 0, width: width, height: height))
        data = bytes
    }
    func rgb(_ x: Int, _ y: Int) -> (Double, Double, Double) {
        let i = (y * width + x) * 4
        return (Double(data[i]) / 255, Double(data[i + 1]) / 255, Double(data[i + 2]) / 255)
    }
}

let args = CommandLine.arguments
guard args.count == 3 else { print("usage: swift Tools/MeasureContrast.swift <capture-dir> <pairs.json>"); exit(2) }
let dir = URL(fileURLWithPath: args[1])
let pairs = try JSONDecoder().decode([Pair].self, from: Data(contentsOf: URL(fileURLWithPath: args[2])))
let files = try FileManager.default.contentsOfDirectory(at: dir, includingPropertiesForKeys: nil)
let scale = 3.0

print("| Screen | Element | Kind | Yardstick | Floor | Worst | Frame | Margin |")
print("|---|---|---|---|---|---|---|---|")
for pair in pairs {
    guard let framesFile = files.first(where: { $0.lastPathComponent.hasSuffix("-\(pair.screen)-frames.json") }),
          let frames = try JSONSerialization.jsonObject(with: Data(contentsOf: framesFile)) as? [String: [Double]],
          let box = frames[pair.element] else {
        print("| \(pair.screen) | \(pair.element) | — | — | \(pair.floor) | **not found** | — | — |"); continue
    }
    let others = frames.filter { $0.key != pair.element }.map(\.value)
    let (inner, outer) = pair.yardstick == "beside" ? (2.0, 6.0) : (10.0, 14.0)
    var worst = (value: Double.infinity, frame: "")
    for shot in files where shot.pathExtension == "png" && shot.lastPathComponent.contains("-\(pair.screen)-") {
        guard let px = Pixels(shot) else { continue }
        func inside(_ b: [Double], _ x: Int, _ y: Int, pad: Double) -> Bool {
            let (bx, by) = ((Double(x) / scale), (Double(y) / scale))
            return bx >= b[0] - pad && bx <= b[0] + b[2] + pad && by >= b[1] - pad && by <= b[1] + b[3] + pad
        }
        // The ground: the ring between `inner` and `outer` around the box, every other element masked (3 px).
        var ground = 0.0
        let x0 = max(0, Int((box[0] - outer) * scale)), x1 = min(px.width - 1, Int((box[0] + box[2] + outer) * scale))
        let y0 = max(0, Int((box[1] - outer) * scale)), y1 = min(px.height - 1, Int((box[1] + box[3] + outer) * scale))
        for y in y0...y1 { for x in x0...x1 where !inside(box, x, y, pad: inner)
            && !others.contains(where: { inside($0, x, y, pad: 1) }) {
            let (r, g, b) = px.rgb(x, y)
            ground = max(ground, luminance(r, g, b))
        } }
        // The ink: white at its opacity over the lightest ground, or the mark's own 98th-percentile pixel.
        let ink: Double
        if pair.ink == "white" {
            let a = pair.inkOpacity ?? 1
            // White composited over the lightest ground, channel by channel, then its luminance.
            var lightest = (0.0, 0.0, 0.0), best = -1.0
            for y in y0...y1 { for x in x0...x1 where !inside(box, x, y, pad: inner) {
                let c = px.rgb(x, y); let l = luminance(c.0, c.1, c.2)
                if l > best && l <= ground { best = l; lightest = c }
            } }
            ink = luminance(a + (1 - a) * lightest.0, a + (1 - a) * lightest.1, a + (1 - a) * lightest.2)
        } else {
            var values: [Double] = []
            for y in Int(box[1] * scale)..<min(px.height, Int((box[1] + box[3]) * scale)) {
                for x in Int(box[0] * scale)..<min(px.width, Int((box[0] + box[2]) * scale)) {
                    let (r, g, b) = px.rgb(x, y); values.append(luminance(r, g, b))
                }
            }
            values.sort()
            ink = values.isEmpty ? 0 : values[Int(Double(values.count - 1) * 0.98)]
        }
        let figure = ratio(ink, ground)
        if figure < worst.value { worst = (figure, shot.lastPathComponent) }
    }
    let margin = worst.value - pair.floor
    print("| \(pair.screen) | \(pair.element) | \(pair.kind) | \(pair.yardstick) | \(pair.floor) | \(String(format: "%.2f", worst.value)) | \(worst.frame) | \(String(format: "%+.2f", margin)) |")
}
```

//
//  AppIconSwitcher.swift

import UIKit

/// iOS's app icon, reached through one seam (spec `2026-11-22-set-c-design.md` §3.7; rule `10-architecture`).
///
/// **Nothing is stored**: iOS keeps the choice, and the card reads it back each time it appears. A struct of
/// closures built by ``live()``, each reaching `UIApplication.shared` in its own body — `ReminderScheduler`'s shape —
/// so no UIKit value sits in a `static let` (rule `43-concurrency`). iOS shows its own alert after a change; a
/// failed change leaves the icon and the selection as they were, said under `#if DEBUG`, never retried.
struct AppIconSwitcher {
    var current: @MainActor () -> String?
    var set: @MainActor (String?) async throws -> Void

    static func live() -> AppIconSwitcher {
        AppIconSwitcher(
            current: { UIApplication.shared.alternateIconName },
            set: { try await UIApplication.shared.setAlternateIconName($0) }
        )
    }
}

extension AppTheme {
    /// The alternate icon set's name in the asset catalogue — `nil` for Aurora's, the primary (D4, D21).
    nonisolated var iconName: String? { self == .aurora ? nil : "AppIcon-\(rawValue.capitalized)" }

    /// The picker's image of the icon, generated beside it by `Tools/GenerateAppIcon.swift`.
    nonisolated var iconPreviewName: String { "IconPreview-\(rawValue.capitalized)" }

    /// The theme whose icon iOS shows, read from `alternateIconName`; Aurora's for the primary and for a name this
    /// build does not have.
    nonisolated static func icon(named name: String?) -> AppTheme {
        allCases.first { $0.iconName == name } ?? .aurora
    }
}
