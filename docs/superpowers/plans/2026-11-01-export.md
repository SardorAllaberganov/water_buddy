# Export — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan
> task-by-task, in the main loop — `CLAUDE.md` has **no implementation subagent** (*Implementation — one
> loop, tests first*). Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a Premium holder export every serving as a CSV file from Settings, through the share sheet
(roadmap 1.4.4).

**Architecture:** `ServingsExport`, a pure app-only value, turns plain rows into the file and a summary, and
answers *nothing to export* and *the read failed* as distinct outcomes. Settings' new card reads
`manager.logs(since: nil)` once, on the tap, and opens `ExportSheet`, which shows the summary and hands the
file to `ShareLink` in memory. `DataManager` does not change, so the widget and the watch are untouched.

**Tech Stack:** Swift 5 mode, SwiftUI, `CoreTransferable`, `UniformTypeIdentifiers`, swift-testing, XCTest UI
tests. No new entitlement, usage key, stored key or target membership.

**Spec:** `docs/superpowers/specs/2026-11-01-export-design.md` — approved 2026-11-01, D1–D3 the owner's, D4–D15
as recommended, §6's refinements from this plan. Read it before Task 1.

## Review Focus

1. **A serving added later to an earlier day** — History's `+` inserts an old instant after newer rows. It must
   stand in its place by date, not by when it was typed. Pinned in Task 3
   (`aServingAddedLaterToAnEarlierDayStandsInItsPlace`).
2. **A long history** — years of servings must arrive whole, first and last rows included. Pinned in Task 3
   (`aLongHistoryLosesNoRow`, 15,000 rows); the tap's feel is read on the walk.
3. **The edges of a day** — a serving at exactly midnight belongs to the day it starts, and 23:59:59 to the day it
   ends. Pinned in Task 3 (`aServingAtMidnightBelongsToTheDayItStarts`).
4. **The largest text size at the medium detent** — the share button must not be pushed out of reach. Pinned in
   Task 6 (`testTheShareButtonIsReachableAtTheLargestTextSize`).
5. **A Russian holder's summary** — the count takes its Russian plural form and is grouped, while the file's
   bytes stay English and ungrouped. Pinned in Task 4 (`theCountFollowsTheAppsLanguageAndIsGrouped`) and Task 3
   (`theFileDoesNotFollowACalendarsLocale`).

```python
import json, sys
path, spec = sys.argv[1], sys.argv[2]
src = open(path, encoding="utf-8").read()
cat = json.loads(src)
assert json.dumps(cat, indent=2, ensure_ascii=False) + "\n" == src, "the catalogue does not round-trip; stop"

for key, langs in new.items():
    assert key not in cat["strings"], f"{key!r} already exists"
    locs = {lang: {"stringUnit": {"state": "translated", "value": langs[lang]}} for lang in ("en", "ru", "uz")}

open(path, "w", encoding="utf-8").write(json.dumps(cat, indent=2, ensure_ascii=False) + "\n")
print(f"added {len(new)} keys")
```

```swift
//
//  ServingsExportTests.swift
//  WaterBuddyTests
//
//  Created by Sardor Allaberganov on 01/11/26.
//

import Foundation
import Testing
@testable import WaterBuddy

/// The export's file, exactly (spec `2026-11-01-export-design.md`). Not `@MainActor`: a canary that
/// `ServingsExport` stays `nonisolated` and imports `Foundation` alone (rule `43-concurrency`).
struct ServingsExportTests {

    private func calendar(_ zone: String, locale: String = "en_US_POSIX") throws -> Calendar {
        var calendar = Calendar(identifier: .gregorian)
        calendar.timeZone = try #require(TimeZone(identifier: zone))
        calendar.locale = Locale(identifier: locale)
        return calendar
    }

    private func instant(_ calendar: Calendar, _ year: Int, _ month: Int, _ day: Int,
                         _ hour: Int = 12, _ minute: Int = 0, _ second: Int = 0) throws -> Date {
        try #require(calendar.date(from: DateComponents(year: year, month: month, day: day,
                                                        hour: hour, minute: minute, second: second)))
    }

    private func file(_ rows: [ServingsExport.Serving], today: Date, calendar: Calendar) throws -> ServingsExport {
        guard case .file(let export) = ServingsExport.outcome(of: rows, today: today, calendar: calendar) else {
            Issue.record("no file")
            throw CancellationError()
        }
        return export
    }

    private func text(_ export: ServingsExport) -> String {
        String(decoding: export.data, as: UTF8.self)
    }

    @Test func theWholeFileIsItsHeadingsAndItsRows() throws {
        let utc = try calendar("UTC")
        let rows = [
            ServingsExport.Serving(amount: 250, timestamp: try instant(utc, 2026, 3, 3, 9, 14)),
            ServingsExport.Serving(amount: 500, timestamp: try instant(utc, 2026, 3, 3, 13, 2)),
        ]
        let export = try file(rows, today: try instant(utc, 2026, 11, 1), calendar: utc)
        #expect(text(export) == "date,time,amount_ml\r\n2026-03-03,09:14,250\r\n2026-03-03,13:02,500\r\n")
    }

    /// `logs(since:)` answers newest first.
    @Test func rowsRunOldestFirst() throws {
        let utc = try calendar("UTC")
        let rows = [
            ServingsExport.Serving(amount: 500, timestamp: try instant(utc, 2026, 3, 4, 8)),
            ServingsExport.Serving(amount: 250, timestamp: try instant(utc, 2026, 3, 3, 8)),
        ]
        let lines = text(try file(rows, today: try instant(utc, 2026, 11, 1), calendar: utc))
            .components(separatedBy: "\r\n")
        #expect(lines[1] == "2026-03-03,08:00,250")
        #expect(lines[2] == "2026-03-04,08:00,500")
    }

    /// History's `+` logs an old instant after newer rows, so the order is the instant's, never the input's.
    @Test func aServingAddedLaterToAnEarlierDayStandsInItsPlace() throws {
        let utc = try calendar("UTC")
        let rows = [
            ServingsExport.Serving(amount: 200, timestamp: try instant(utc, 2026, 3, 5, 8)),
            ServingsExport.Serving(amount: 100, timestamp: try instant(utc, 2026, 3, 1, 8)),
            ServingsExport.Serving(amount: 300, timestamp: try instant(utc, 2026, 3, 9, 8)),
        ]
        let lines = text(try file(rows, today: try instant(utc, 2026, 11, 1), calendar: utc))
            .components(separatedBy: "\r\n")
        #expect(Array(lines[1...3]) == ["2026-03-01,08:00,100", "2026-03-05,08:00,200", "2026-03-09,08:00,300"])
    }

    /// As History reads it: the wall clock in the zone the phone is in now (D6).
    @Test func aRowReadsItsInstantInTheCalendarsZone() throws {
        let utc = try calendar("UTC")
        let moment = try instant(utc, 2026, 6, 15, 3, 30)
        let rows = [ServingsExport.Serving(amount: 250, timestamp: moment)]
        let paris = try calendar("Europe/Paris"), angeles = try calendar("America/Los_Angeles")
        #expect(text(try file(rows, today: moment, calendar: paris)).contains("2026-06-15,05:30,250"))
        #expect(text(try file(rows, today: moment, calendar: angeles)).contains("2026-06-14,20:30,250"))
    }

    /// The clocks go back on 1 November 2026 in Los Angeles: 01:30 happens twice, an hour apart.
    @Test func bothRowsOfARepeatedHourAreKept() throws {
        let utc = try calendar("UTC"), angeles = try calendar("America/Los_Angeles")
        let rows = [
            ServingsExport.Serving(amount: 250, timestamp: try instant(utc, 2026, 11, 1, 8, 30)),
            ServingsExport.Serving(amount: 500, timestamp: try instant(utc, 2026, 11, 1, 9, 30)),
        ]
        let lines = text(try file(rows, today: try instant(utc, 2026, 11, 1, 20), calendar: angeles))
            .components(separatedBy: "\r\n")
        #expect(lines[1] == "2026-11-01,01:30,250")
        #expect(lines[2] == "2026-11-01,01:30,500")
    }

    @Test func aServingAtMidnightBelongsToTheDayItStarts() throws {
        let utc = try calendar("UTC")
        let rows = [
            ServingsExport.Serving(amount: 100, timestamp: try instant(utc, 2026, 3, 3, 23, 59, 59)),
            ServingsExport.Serving(amount: 200, timestamp: try instant(utc, 2026, 3, 4, 0, 0, 0)),
        ]
        let lines = text(try file(rows, today: try instant(utc, 2026, 11, 1), calendar: utc))
            .components(separatedBy: "\r\n")
        #expect(lines[1] == "2026-03-03,23:59,100")
        #expect(lines[2] == "2026-03-04,00:00,200")
    }

    @Test func twoServingsInOneMinuteStayTwoRows() throws {
        let utc = try calendar("UTC")
        let moment = try instant(utc, 2026, 3, 3, 9, 14)
        let rows = [
            ServingsExport.Serving(amount: 250, timestamp: moment),
            ServingsExport.Serving(amount: 250, timestamp: moment.addingTimeInterval(20)),
        ]
        let export = try file(rows, today: moment, calendar: utc)
        #expect(export.count == 2)
        #expect(text(export).components(separatedBy: "\r\n").count == 4)
    }

    /// Never clamped and never grouped: the file is the record (D7). Nothing here sums, so nothing can overflow.
    @Test func anAmountIsWrittenAsStored() throws {
        let utc = try calendar("UTC")
        let moment = try instant(utc, 2026, 3, 3)
        let rows = [
            ServingsExport.Serving(amount: 100_000, timestamp: moment),
            ServingsExport.Serving(amount: Int.max, timestamp: moment.addingTimeInterval(60)),
        ]
        let body = text(try file(rows, today: moment, calendar: utc))
        #expect(body.contains(",100000\r\n"))
        #expect(body.contains(",\(Int.max)\r\n"))
    }

    /// 02:00 UTC on the 1st is still the 31st in Los Angeles.
    @Test func theNameCarriesTodaysDayOnTheCalendar() throws {
        let utc = try calendar("UTC"), angeles = try calendar("America/Los_Angeles")
        let today = try instant(utc, 2026, 11, 1, 2)
        let rows = [ServingsExport.Serving(amount: 250, timestamp: today)]
        #expect(try file(rows, today: today, calendar: utc).name == "WaterBuddy-2026-11-01.csv")
        #expect(try file(rows, today: today, calendar: angeles).name == "WaterBuddy-2026-10-31.csv")
    }

    @Test func theSummaryCountsTheFile() throws {
        let utc = try calendar("UTC")
        let early = try instant(utc, 2026, 3, 3), middle = try instant(utc, 2026, 4, 4), late = try instant(utc, 2026, 5, 5)
        let rows = [late, early, middle].map { ServingsExport.Serving(amount: 250, timestamp: $0) }
        let export = try file(rows, today: late, calendar: utc)
        #expect(export.count == 3)
        #expect(export.first == early)
        #expect(export.last == late)
    }

    @Test func noRowsIsNothingToExport() throws {
        let utc = try calendar("UTC")
        #expect(ServingsExport.outcome(of: [], today: .distantPast, calendar: utc) == .nothing)
```

```swift
//
//  ServingsExport.swift
//  WaterBuddy
//
//  Created by Sardor Allaberganov on 01/11/26.
//

import Foundation

/// Every serving, as the CSV file Premium's export sends, and what the sheet says of it (spec
/// `2026-11-01-export-design.md`).
///
/// **A pure value, as ``HealthPlan`` is.** It is handed plain rows and imports nothing but Foundation, so
/// every case is testable without a store or a screen, and nothing past the read holds a ``WaterLog``.
///
/// **No locale and no bundle, on purpose.** The headings are fixed English in every language, a format and
/// not copy (D3), and a date is a calendar's components, never a date style, so the file's bytes cannot
/// follow the app's language. Three columns and nothing else — rule `70-privacy`'s *The export file*.
///
/// A row reads its instant on the calendar it is handed, the zone the phone is in now, as History does:
/// ``WaterLog`` keeps no zone (D6).
///
/// `nonisolated`, stated outright: the app's default isolation is the main actor, and
/// `ServingsExportTests` reads this from off it, as a canary (rule `43-concurrency`).
nonisolated struct ServingsExport: Identifiable, Sendable, Equatable {

    /// One row, as the store holds it.
    nonisolated struct Serving: Sendable, Equatable {
        let amount: Int
        let timestamp: Date
    }

    /// What one read came to. A failed read stays apart from an empty history: it is never answered with a
    /// file of headings (rule `20-state`).
    nonisolated enum Outcome: Sendable, Equatable {
        case file(ServingsExport)
        case nothing
        case unreadable
    }

    /// *WaterBuddy-2026-11-01.csv*: dated, so two exports in Files do not collide.
    let name: String
    /// The file itself, in memory: the app writes none (rule `70-privacy`).
    let data: Data
    /// How many servings it holds.
    let count: Int
    /// The oldest serving's instant.
    let first: Date
    /// The newest serving's instant.
    let last: Date

    var id: String { "\(name).\(count)" }

    private static let headings = "date,time,amount_ml"
    /// RFC 4180's line ending.
    private static let lineEnding = "\r\n"

    /// The answer for one read. `nil` rows is a failed read; `today` dates the file's name.
    static func outcome(of rows: [Serving]?, today: Date, calendar: Calendar) -> Outcome {
        guard let rows else { return .unreadable }
        // By instant, not by the read's order: History's `+` logs an old instant after newer rows.
        let ordered = rows.sorted { $0.timestamp < $1.timestamp }
        guard let first = ordered.first, let last = ordered.last else { return .nothing }
        let lines = [headings] + ordered.map { line(for: $0, calendar: calendar) }
        return .file(ServingsExport(
            name: "WaterBuddy-\(day(of: today, calendar: calendar)).csv",
            data: Data((lines.joined(separator: lineEnding) + lineEnding).utf8),
            count: ordered.count,
            first: first.timestamp,
            last: last.timestamp
        ))
    }

    /// No field can hold a comma, so nothing is quoted.
    private static func line(for serving: Serving, calendar: Calendar) -> String {
        let clock = calendar.dateComponents([.hour, .minute], from: serving.timestamp)
        let time = String(format: "%02d:%02d", clock.hour ?? 0, clock.minute ?? 0)
        return "\(day(of: serving.timestamp, calendar: calendar)),\(time),\(serving.amount)"
    }

    private static func day(of date: Date, calendar: Calendar) -> String {
        let day = calendar.dateComponents([.year, .month, .day], from: date)
        return String(format: "%04d-%02d-%02d", day.year ?? 0, day.month ?? 0, day.day ?? 0)
    }
}
```
